// Floating icons positioned around the entire hero section
function FloatingIcon({
  icon: Icon,
  color,
  delay,
  x,
  y,
}: {
  icon: React.ElementType;
  color: string;
  delay: number;
  x: number;
  y: number;
}) {
  const randomDelay = useMemo(
    () => delay + ((delay * 1000) % 500) / 1000,
    [delay]
  );

  return (
    <motion.div
      className="absolute z-10 p-3 bg-background/90 backdrop-blur-md border rounded-xl shadow-lg flex items-center justify-center"
      initial={{ opacity: 0, scale: 0 }}
      animate={{
        opacity: 1,
        scale: 1,
        y: [0, -15, 0],
      }}
      transition={{
        opacity: { delay, duration: 0.5 },
        scale: { delay, duration: 0.5 },
        y: {
          duration: 4,
          repeat: Infinity,
          ease: "easeInOut",
          delay: randomDelay,
        },
      }}
      whileHover={{ scale: 1.15, rotate: 5 }}
      style={{
        left: `${x}%`,
        top: `${y}%`,
        borderColor: `${color}30`,
        boxShadow: `0 8px 32px -4px ${color}25`,
      }}
    >
      <Icon className="w-6 h-6" style={{ color }} />
    </motion.div>
  );
}
